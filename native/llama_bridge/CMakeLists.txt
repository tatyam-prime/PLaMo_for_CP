cmake_minimum_required(VERSION 3.20)
project(cp_llama_bridge LANGUAGES CXX)

option(CP_LLAMA_BRIDGE_TESTS "Build model-free tests with an in-memory fake backend" OFF)

if(CP_LLAMA_BRIDGE_TESTS)
  enable_testing()
  add_executable(cp_llama_bridge_test bridge.cpp tests/fake_runtime.cpp tests/bridge_test.cpp)
  target_compile_features(cp_llama_bridge_test PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(cp_llama_bridge_test PRIVATE /utf-8 /UNDEBUG)
  else()
    target_compile_options(cp_llama_bridge_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME bridge_model_free COMMAND cp_llama_bridge_test "${CMAKE_CURRENT_BINARY_DIR}")
else()
  if(NOT WIN32)
    message(FATAL_ERROR "Build the production bridge on Windows; no runtime is downloaded by this project.")
  endif()
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The production bridge requires a 64-bit build.")
  endif()
  add_library(cp_llama_bridge SHARED bridge.cpp runtime_windows.cpp)
  target_compile_features(cp_llama_bridge PRIVATE cxx_std_17)
  if(MSVC)
    # No C++/CRT object crosses our ABI; a static CRT avoids a new user dependency.
    set_property(TARGET cp_llama_bridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(cp_llama_bridge PRIVATE /W4 /EHsc /utf-8)
  endif()
  set_target_properties(cp_llama_bridge PROPERTIES PREFIX "" OUTPUT_NAME "cp_llama_bridge")
endif()
